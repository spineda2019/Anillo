# See the README for more context around syntax of this grammar (not Anillo
# itself), such as how whitespace is handled, and how + and * are intended to
# be interpretted

# An "Anillo" object in its entirety can be considered as a collection of one
# or more "Ingots", a cheeky play on words.
Root             := Ingot* eof

Ingot            := ExternalFunction | ISR

ExternalFunction := SKIP_WHITE KEYWORD_EXTERN FunctionDecl RingLevel?

FunctionDecl     := Identifier LEFT_PAREN ArgList? RIGHT_PAREN

Identifier       := SKIP_WHITE [A-Za-z_][A-Za-z0-9_]* SKIP_WHITE

ArgList          := SKIP_WHITE CType SKIP_WHITE Identifier RemainingList*
RemainingList    := SKIP_WHITE COMMA ArgList

CType            := U8 | I8 | U16 | I16 | U32 | I32 | U64 | I64

RingLevel        := KEYWORD_RING_LEVEL LEFT_PAREN RingName RIGHT_PAREN

RingName         := KEYWORD_USER | KEYWORD_SUPER

ISR              := SKIP_WHITE KEYWORD_ISR Identifier IsrID RingLevel? IsrBody

IsrID            := SKIP_WHITE LEFT_PAREN SKIP_WHITE [0-255] SKIP_WHITE RIGHT_PAREN SKIP_WHITE

IsrBody          := SKIP_WHITE LEFT_BRACKET SKIP_WHITE CallExpression*

CallExpression   := KEYWORD_CALL Identifier LEFT_PAREN ArgValList* RIGHT_PAREN

ArgValList       := Identifier SKIP_WHITE RemainingAVList*
RemainingAVList  := COMMA SKIP_WHITE ArgValList


# Keywords and tokens
KEYWORD_EXTERN     := 'extern'    SKIP_WHITE
KEYWORD_RING_LEVEL := 'WithLevel' SKIP_WHITE
KEYWORD_USER       := 'User'      SKIP_WHITE
KEYWORD_SUPER      := 'Super'     SKIP_WHITE
KEYWORD_ISR        := 'isr'       SKIP_WHITE
KEYWORD_CALL       := 'call'      SKIP_WHITE
LEFT_PAREN         := '('
LEFT_BRACKET       := '{'
RIGHT_PAREN        := ')'
RIGHT_BRACKET      := '}'
COMMA              := ','
SKIP_WHITE         := ... # We can be exhaustive if we'd like...
